CREATE TABLE Transactions (
    TransactionID INT PRIMARY KEY,
    AccountNo BIGINT,
    Amount DECIMAL(12,2),
    Type VARCHAR(20),
    TransactionDate DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE AuditLog (
    AuditID INT AUTO_INCREMENT PRIMARY KEY,
    Event VARCHAR(100),
    EventDate DATETIME
);
