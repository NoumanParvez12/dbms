create or replace procedure calcBonus
IS 
  Cursor emp_cursor IS
    select EmpID, Salary from Employee;
  Begin
    for emp_rec in emp_cursor loop

      update employee set bonus = emp_rec.Salary*0.10
      where EmpID = emp_rec.EmpID;

    END LOOP;

  END;
/

create or replace function GetDeptCount(dept_name Varchar2)
return NUMBER
IS
  dept_count NUMBER;
  Begin
    select count(*) into dept_count from employee
    where Department = dept_name;

    return dept_count;
  END;

CREATE PROCEDURE fine()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE id INT;
    DECLARE mname VARCHAR(30);
    DECLARE bname VARCHAR(50);
    DECLARE idate DATE;
    DECLARE days INT;
    DECLARE f INT;

    DECLARE issue_cur CURSOR FOR
        SELECT issue_id, member_name, book_name, issue_date
        FROM issue;

    DECLARE CONTINUE HANDLER FOR NOT FOUND
        SET done = 1;

    OPEN issue_cur;

    read_loop: LOOP

        FETCH issue_cur INTO id, mname, bname, idate;

        IF done THEN
            LEAVE read_loop;
        END IF;

        SET days = DATEDIFF(CURDATE(), idate);

        IF days > 0 THEN

            SET f = days * 10;

            UPDATE issue
            SET fine = f
            WHERE issue_id = id;

            SELECT mname AS Member_Name,
                   bname AS Book_Name,
                   days AS Delayed_Days,
                   f AS Fine_Amount;

        ELSE

            UPDATE issue
            SET fine = 0
            WHERE issue_id = id;

        END IF;

    END LOOP;

    CLOSE issue_cur;

END //

DELIMITER ;

CALL fine();
